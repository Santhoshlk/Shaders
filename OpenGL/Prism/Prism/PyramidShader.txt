#shader vertex

#version 460 core

layout(location = 0) in vec3 pos;
layout(location = 1) in vec3 color;

out vec3 v_color;
uniform mat4 u_mvp;

void main()
{
   gl_Position = u_mvp * vec4(pos,1.f);
   v_color = color;
}


#shader fragment
#version 460 core

in vec3 v_color;
out vec4 Color;

void main()
{
    Color = vec4(v_color,1.f);
}
