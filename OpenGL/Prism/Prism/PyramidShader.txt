#shader vertex

#version 460 core

layout(location = 0) in vec3 pos;
layout(location = 1) in vec3 color;

out vec3 v_color;
uniform mat4 u_view;
uniform mat4 u_proj;


void main()
{
   vec4 result = u_proj * u_view * vec4(pos,1.f);

   vec4 positions = vec4(1.f);
    positions.x = result.x/result.w;
    positions.y = result.y/result.w;
    positions.z = result.z/result.w;
   gl_Position = positions;
   v_color = color;
}


#shader fragment
#version 460 core

in vec3 v_color;
out vec4 Color;

void main()
{
    Color = vec4(v_color,1.f);
}
