#shader vertex 
#version 460 core


layout(location = 0) in vec2 Pos;
layout(location = 1) in vec2 texcoord;

out vec2 v_texcoord;

void main()
{
   gl_Position = vec4(vec3(Pos,0.f),1.f);
   v_texcoord = texcoord;
}

#shader fragment
#version 460 core


uniform sampler2D u_TexSlot;
in vec2 v_texcoord;
layout(location = 0)out vec4 Color;
void main()
{
   Color = texture(u_TexSlot,v_texcoord);
   
}
