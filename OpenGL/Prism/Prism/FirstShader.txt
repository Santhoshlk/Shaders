#shader vertex 
#version 460 core


layout(location = 3) in vec2 Pos;
layout(location = 4) in vec3 vcolor;

out vec3 fcolor;

void main()
{
   gl_Position = vec4(vec3(Pos,0.f),1.f);
   fcolor = vcolor;
}

#shader fragment
#version 460 core

in vec3 fcolor;
uniform float u_alpha;
layout(location = 0)out vec4 Color;

void main()
{
   Color = vec4(fcolor,u_alpha);
}
